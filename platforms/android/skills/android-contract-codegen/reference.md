## Mode A — Codegen commands and wiring

### A1. OpenAPI → Retrofit2 (openapi-generator Gradle plugin)

`build.gradle.kts` (`:network` module):

```kotlin
plugins {
    id("org.openapi.generator") version "7.6.0"
    id("org.jetbrains.kotlin.plugin.serialization") version "1.9.23"
}

openApiGenerate {
    generatorName.set("kotlin")
    inputSpec.set("$rootDir/specs/openapi.yaml")   // single source of truth
    outputDir.set("$buildDir/generated/openapi")
    apiPackage.set("com.example.network.api")
    modelPackage.set("com.example.network.model")
    configOptions.set(mapOf(
        "library"              to "jvm-retrofit2",
        "serializationLibrary" to "kotlinx_serialization",
        "useCoroutines"        to "true",
        "enumPropertyNaming"   to "UPPERCASE",
        "nullableReturnType"   to "true",
    ))
}

// Make generated sources available to compilation
kotlin.sourceSets["main"].kotlin.srcDir("$buildDir/generated/openapi/src/main/kotlin")
```

Regenerate whenever the spec changes:

```bash
./gradlew openApiGenerate
# Review the diff in build/generated/openapi/ before committing
git diff build/generated/openapi/
```

### A2. GraphQL → Apollo Kotlin

`build.gradle.kts` (`:network` module):

```kotlin
plugins {
    id("com.apollographql.apollo3") version "3.8.5"
}

apollo {
    service("api") {
        packageName.set("com.example.network.graphql")
        schemaFile.set(file("src/main/graphql/schema.graphqls"))  // single source of truth
        srcDir("src/main/graphql")                                 // .graphql operation files
    }
}
```

```bash
./gradlew generateApolloSources
# Review: src/generated/
```

### A3. Wiring a generated Retrofit call with sealed-class error handling

```kotlin
// Sealed result type — covers all 3 outcomes the spec declares
sealed class ApiResult<out T> {
    data class Success<T>(val data: T)                          : ApiResult<T>()
    data class HttpError(val code: Int, val body: ApiErrorBody) : ApiResult<Nothing>()
    data class NetworkError(val cause: Throwable)               : ApiResult<Nothing>()
}

// Generated by openapi-generator — do not edit
interface UserApi {
    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: String): UserResponse
}

// Repository — wraps generated call in ApiResult
class UserRepository @Inject constructor(private val api: UserApi) {

    suspend fun getUser(id: String): ApiResult<UserResponse> = runCatching {
        api.getUser(id)
    }.fold(
        onSuccess = { ApiResult.Success(it) },
        onFailure = { e ->
            when (e) {
                is HttpException -> {
                    val errBody = e.response()?.errorBody()?.string()
                        ?.let { Json.decodeFromString<ApiErrorBody>(it) }
                        ?: ApiErrorBody(message = "Unknown error")
                    ApiResult.HttpError(e.code(), errBody)
                }
                else -> ApiResult.NetworkError(e)
            }
        }
    )
}

// ViewModel — exhaustive when; no else needed
viewModelScope.launch {
    when (val result = repo.getUser(id)) {
        is ApiResult.Success      -> _state.update { it.copy(user = result.data) }
        is ApiResult.HttpError    -> _state.update { it.copy(error = "HTTP ${result.code}: ${result.body.message}") }
        is ApiResult.NetworkError -> _state.update { it.copy(error = "Network error") }
    }
}
```

### A4. Wiring a generated Apollo Kotlin call

```kotlin
// Operation file: src/main/graphql/GetUser.graphql
// query GetUser($id: ID!) { user(id: $id) { id name email status } }

class UserRepository @Inject constructor(private val apollo: ApolloClient) {

    suspend fun getUser(id: String): ApiResult<GetUserQuery.User> {
        val response = apollo.query(GetUserQuery(id = id)).execute()
        return when {
            response.hasErrors()  -> ApiResult.HttpError(422, ApiErrorBody(response.errors!!.first().message))
            response.data != null -> ApiResult.Success(response.data!!.user!!)
            else                  -> ApiResult.NetworkError(IllegalStateException("Empty data"))
        }
    }
}
```

## Rule 1 — Hand-written DTO instead of generated

**Bad** — hand-written data class that duplicates a schema component:

```kotlin
// hand-written in feature/user/UserApi.kt — will drift from the schema
data class UserResponse(
    val id: String,
    val name: String,
    val email: String,
)

interface UserApi {
    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: String): UserResponse
}
```

**Good** — use the class generated from the schema:

```kotlin
// Generated by openapi-generator — build/generated/openapi/…/model/UserResponse.kt
// Do not write this by hand; run ./gradlew openApiGenerate

// In feature/user/UserApi.kt — import and use the generated type
import com.example.network.model.UserResponse

interface UserApi {
    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: String): UserResponse
}
```

## Rule 2 — Untyped API boundary

**Bad** — `Map<String, Any>` at the network boundary:

```kotlin
interface UserApi {
    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: String): Map<String, Any>  // untyped
}

// Caller — stringly-typed access, will crash on type mismatch
val name = (response["name"] as? String) ?: ""
```

**Good** — generated typed model:

```kotlin
// Generated model — all fields typed, nullable matches the schema
import com.example.network.model.UserResponse

interface UserApi {
    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: String): UserResponse
}

// Caller — compile-time safe
val name = response.name   // String or String? per schema
```

## Rule 3 — Stringly-typed endpoint

**Bad** — raw `@GET` annotation duplicating the spec path:

```kotlin
// Duplicates the spec; path can drift without any compilation error
interface UserApi {
    @GET("users/{userId}")          // spec says "users/{id}"
    suspend fun fetchUser(@Path("userId") id: String): UserResponse
}
```

**Good** — use the generated API interface verbatim:

```kotlin
// Generated by openapi-generator — the path comes from the spec, not hand-written
// com.example.network.api.UserApi (generated)
//   @GET("users/{id}")
//   suspend fun getUser(@Path("id") id: String): UserResponse

// Provide via DI and call directly — no re-declaration needed
class UserRepository @Inject constructor(private val api: UserApi) {
    suspend fun getUser(id: String) = api.getUser(id)   // path owned by the generator
}
```

## Rule 4 — Unmodeled error/status responses

**Bad** — only the 200 path; errors silently swallowed:

```kotlin
suspend fun getUser(id: String): UserResponse? = try {
    api.getUser(id)
} catch (e: Exception) {
    null   // 404, 401, 503 all become null — caller has no typed error info
}
```

**Good** — sealed class covers all declared responses:

```kotlin
// ApiErrorBody is also generated from the schema's error component
sealed class ApiResult<out T> {
    data class Success<T>(val data: T)                          : ApiResult<T>()
    data class HttpError(val code: Int, val body: ApiErrorBody) : ApiResult<Nothing>()
    data class NetworkError(val cause: Throwable)               : ApiResult<Nothing>()
}

suspend fun getUser(id: String): ApiResult<UserResponse> = runCatching {
    api.getUser(id)
}.fold(
    onSuccess  = { ApiResult.Success(it) },
    onFailure  = { e ->
        when (e) {
            is HttpException -> ApiResult.HttpError(
                e.code(),
                e.response()?.errorBody()?.string()
                    ?.let { Json.decodeFromString<ApiErrorBody>(it) }
                    ?: ApiErrorBody("Unknown")
            )
            else -> ApiResult.NetworkError(e)
        }
    }
)
```

## Rule 5 — Manual (de)serialization drift

**Bad** — hand-rolled JSON access; snake/camelCase mismatch:

```kotlin
val obj = JSONObject(responseBody)
val firstName = obj.getString("first_name")   // schema says "firstName" — runtime crash
val age = obj.getInt("age")
```

**Good** — kotlinx.serialization with `@SerialName` generated by the tool:

```kotlin
// Generated model — serialization handled by the codec, not by hand
@Serializable
data class UserResponse(
    @SerialName("firstName") val firstName: String,
    @SerialName("age")       val age: Int,
)

// Retrofit converter factory wires the codec — no manual parsing needed
val retrofit = Retrofit.Builder()
    .addConverterFactory(Json.asConverterFactory("application/json".toMediaType()))
    .build()
```

## Rule 6 — Nullable/required mismatch

**Bad** — force-unwrap on a field the schema marks optional:

```kotlin
// Schema: UserResponse.middleName is nullable (not in required[])
@Serializable
data class UserResponse(
    val id: String,
    val middleName: String,   // should be String? — serializer throws on null
)

// Caller
val initials = user.middleName!!.first()   // NullPointerException at runtime
```

**Good** — `String?` matches the schema; null-safe access:

```kotlin
// Generated correctly — matches nullable: true in the schema
@Serializable
data class UserResponse(
    val id: String,
    val middleName: String?,  // nullable per schema
)

// Caller — null-safe
val initials = user.middleName?.firstOrNull()
```

## Rule 7 — Enum as raw string

**Bad** — string literal comparison against a schema enum:

```kotlin
// Schema defines UserStatus enum: active | suspended | deleted
val status: String = user.status        // typed as String
if (status == "active") showDashboard() // silent miss if schema adds "pending"
```

**Good** — generated `enum class`; exhaustive `when`:

```kotlin
// Generated by openapi-generator
enum class UserStatus {
    ACTIVE, SUSPENDED, DELETED
}

@Serializable
data class UserResponse(
    val id: String,
    val status: UserStatus,   // generated enum type
)

// Caller — compiler enforces exhaustiveness; no else
when (user.status) {
    UserStatus.ACTIVE    -> showDashboard()
    UserStatus.SUSPENDED -> showSuspendedBanner()
    UserStatus.DELETED   -> navigateToSignIn()
}
```

## Rule 8 — Regen without diffing

**Bad** — silent regen after a spec bump, no diff review:

```bash
# Spec bumped openapi.yaml: renamed "name" → "displayName"
./gradlew openApiGenerate
git add -A && git commit -m "regen"   # no review — callers accessing .name now fail to compile
```

**Good** — review generated diff before committing:

```bash
./gradlew openApiGenerate

# Review what changed in the generated output
git diff build/generated/openapi/

# If a field was renamed, callers show compile errors — fix before committing
# Include the generated diff in the PR so reviewers can verify the surface change
git add build/generated/openapi/ src/
git commit -m "chore: bump spec v2.1 — rename name→displayName, regen client"
```

## Rule 9 — Hand-edited generated file

**Bad** — adding logic directly inside a generated file:

```kotlin
// build/generated/openapi/src/main/kotlin/com/example/network/api/UserApi.kt
// GENERATED — do not edit

interface UserApi {
    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: String): UserResponse

    // ← AI added this helper here — lost on next ./gradlew openApiGenerate
    suspend fun getCurrentUser(): UserResponse = getUser("me")
}
```

**Good** — add extensions outside the generated dir:

```kotlin
// network/src/main/kotlin/com/example/network/ext/UserApiExt.kt  (hand-written, never regenerated)
import com.example.network.api.UserApi

suspend fun UserApi.getCurrentUser(): UserResponse = getUser("me")
```

## Rule 10 — No single source of truth

**Bad** — two independent class definitions for the same schema type:

```kotlin
// Generated: build/generated/openapi/.../model/UserResponse.kt
data class UserResponse(val id: String, val name: String, val email: String)

// Hand-written in feature module — diverges the moment the schema adds a field
data class UserDto(val id: String, val name: String, val email: String)

// Mapper between two classes that represent the same thing
fun UserResponse.toDto() = UserDto(id, name, email)
```

**Good** — one generated class used everywhere; no mapper needed:

```kotlin
// Generated: build/generated/openapi/.../model/UserResponse.kt
// Import and use directly in the feature module

import com.example.network.model.UserResponse

// Repository returns the generated type; ViewModel and UI consume it directly
// A dedicated UI model is fine — but it should map FROM the generated type, not duplicate it
data class UserUiModel(val displayName: String)

fun UserResponse.toUiModel() = UserUiModel(displayName = name)
```

## Contract testing with android-testing

Wire a `MockWebServer` (OkHttp) or Apollo's `MockServer` in your tests to assert that the generated
client correctly serializes requests and deserializes spec-compliant responses. See
[android-testing](../android-testing/SKILL.md) for setup patterns.

```kotlin
@Test
fun `getUser returns Success with correct model`() = runTest {
    mockWebServer.enqueue(
        MockResponse()
            .setBody("""{"id":"1","name":"Alice","status":"ACTIVE"}""")
            .setResponseCode(200)
    )

    val result = repo.getUser("1")

    assertIs<ApiResult.Success<UserResponse>>(result)
    assertEquals("Alice", result.data.name)
    assertEquals(UserStatus.ACTIVE, result.data.status)
}

@Test
fun `getUser returns HttpError on 404`() = runTest {
    mockWebServer.enqueue(
        MockResponse()
            .setBody("""{"message":"not found"}""")
            .setResponseCode(404)
    )

    val result = repo.getUser("missing")

    assertIs<ApiResult.HttpError>(result)
    assertEquals(404, result.code)
}
```
